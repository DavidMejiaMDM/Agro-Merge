                direction TB

                class usuarios {
                    +int id_usuario [PK]
                    +varchar nombre
                    +varchar email
                    +varchar contrasena
                    +varchar foto_perfil
                    +enum rol
                    +enum estado
                    +datetime fecha_registro
                    +varchar codigo_verificacion
                }

                class vendedores {
                    +int id_vendedor [PK]
                    +int id_usuario [FK]
                    +varchar nombre_tienda
                    +text descripcion
                    +varchar telefono
                }

                class categorias {
                    +int id_categoria [PK]
                    +varchar nombre
                    +text descripcion
                }

                class productos {
                    +int id_producto [PK]
                    +int id_vendedor [FK]
                    +int id_categoria [FK]
                    +varchar nombre
                    +text descripcion
                    +decimal precio
                    +int stock
                    +varchar imagen
                    +enum estado
                    +datetime fecha_publicacion
                    +datetime fecha_creacion
                }

                class carrito {
                    +int id_carrito [PK]
                    +int id_usuario [FK]
                    +datetime fecha_creacion
                }

                class detalle_carrito {
                    +int id_detalle_carrito [PK]
                    +int id_carrito [FK]
                    +int id_producto [FK]
                    +int cantidad
                    +decimal precio_unitario
                }

                class pedidos {
                    +int id_pedido [PK]
                    +int id_comprador [FK]
                    +enum estado
                    +decimal total
                    +varchar direccion_entrega
                    +datetime fecha_pedido
                }


                class detalle_pedido {
                    +int id_detalle [PK]
                    +int id_pedido [FK]
                    +int id_producto [FK]
                    +int cantidad
                    +decimal precio_unitario
                }

                class pagos {
                    +int id_pago [PK]
                    +int id_pedido [FK]
                    +varchar metodo_pago
                    +enum estado_pago
                    +decimal monto
                    +datetime fecha_pago
                }

                class notificaciones {
                    +int id_notificacion [PK]
                    +int id_usuario [FK]
                    +text mensaje
                    +varchar tipo
                    +tinyint leida
                    +datetime fecha
                }
